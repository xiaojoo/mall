# 如果kube-proxy使用的是IPVS模式，你需要启用staticARP

kubectl edit configmap -n kube-system kube-proxy
# 设置staticARP为true
mode: "ipvs"
ipvs:
  strictARP: true

# 创建资源
kubectl apply -f https://raw.githubusercontent.com/metallb/metallb/v0.13.5/config/manifests/metallb-native.yaml
